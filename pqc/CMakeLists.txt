cmake_minimum_required(VERSION 3.10)
project(ccx_pqc CXX)
set(CMAKE_CXX_STANDARD 11)
set(RUST_DIR ${CMAKE_CURRENT_SOURCE_DIR}/ccx-pqc)
set(RUST_LIB ${RUST_DIR}/target/release/libccx_pqc.a)
add_custom_command(OUTPUT ${RUST_LIB} COMMAND cargo build --release WORKING_DIRECTORY ${RUST_DIR} VERBATIM)
add_custom_target(rust_pqc DEPENDS ${RUST_LIB})
add_executable(test_pqc test_pqc.cpp)
target_include_directories(test_pqc PRIVATE include)
add_dependencies(test_pqc rust_pqc)
target_link_libraries(test_pqc ${RUST_LIB} pthread dl m)
